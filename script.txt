local function getLocalProfile()
    local name = string.lower(LocalPlayer.Name)
    for uname, data in pairs(PROFILES) do
        if string.lower(uname) == name then

local function tween(obj, props, t, style, dir)
    TweenService:Create(obj,
        TweenInfo.new(t or 0.22, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props):Play()
end
local function corner(parent, r)
    local c = Instance.new("UICorner", parent)
    c.CornerRadius = UDim.new(0, r or 12)
    return c
end
local function stroke(parent, color, thickness)
    local s = Instance.new("UIStroke", parent)
    s.Color = color or P.STROKE
    s.Thickness = thickness or 1
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return s
end
local function gradient(parent, a, b, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new(a, b)
    g.Rotation = rot or 90
    return g
end
local function pop(obj, amount)
    local base = obj.Size
    tween(obj, { Size = UDim2.new(base.X.Scale, base.X.Offset + (amount or 6), base.Y.Scale, base.Y.Offset + (amount or 6)) }, 0.08, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    task.wait(0.09)
    tween(obj, { Size = base }, 0.15, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end

            Val.Text = fmt(v)
            onChange(v)
        end
        Val.Text = fmt(default)

        local draggingS = false
        local activeSliderInput = nil
        local function fromX(x)
            local rel = math.clamp((x - Track.AbsolutePosition.X) / math.max(Track.AbsoluteSize.X, 1), 0, 1)
            local v = minV + (maxV - minV) * rel
            if decimals == 0 then v = math.floor(v + 0.5) end
            apply(v)
        end
        TrackHit.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                draggingS = true; activeSliderInput = input
                fromX(input.Position.X)
                tween(Knob, { Size = UDim2.new(0, 20, 0, 20) }, 0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not draggingS or input ~= activeSliderInput then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                fromX(input.Position.X)
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if not draggingS or input ~= activeSliderInput then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(Knob, { Size = UDim2.new(0, 16, 0, 16) }, 0.15, Enum.EasingStyle.Back)
                draggingS = false; activeSliderInput = nil
            end
        end)
        task.defer(function() apply(default) end)
        return Holder
    end

    local function clearShaders()
        for _, obj in ipairs(Lighting:GetChildren()) do
            if obj:GetAttribute("livyii_shader") then obj:Destroy() end
        end
    end

    local function applyShaders()
        clearShaders()
        if not Config.ShaderEnabled then return end
        local preset = ShaderPresets[Config.ShaderPreset] or ShaderPresets.pudding
        local k = Config.ShaderIntensity

        local atmo = Instance.new("Atmosphere")
        atmo:SetAttribute("livyii_shader", true)
        atmo.Density      = 0.10 + 0.15 * k
        atmo.Offset       = 0.05 + 0.15 * k
        atmo.Color        = preset.atmo
        atmo.Decay        = preset.decay
        atmo.Glare        = 0.55 + 0.40 * k
        atmo.Haze         = 0.02 + 0.10 * k
        atmo.Parent       = Lighting

        local cc = Instance.new("ColorCorrectionEffect")
        cc:SetAttribute("livyii_shader", true)
        cc.Brightness     = 0.08 + 0.12 * k
        cc.Contrast       = 0.15 + 0.15 * k
        cc.Saturation     = 0.35 + 0.35 * k
        cc.TintColor      = preset.tint
        cc.Parent         = Lighting

        local bloom = Instance.new("BloomEffect")
        bloom:SetAttribute("livyii_shader", true)
        bloom.Intensity   = 1.4 + 1.2 * k
        bloom.Size        = 14 + 12 * k
        bloom.Threshold   = 0.55 - 0.15 * k
        bloom.Parent      = Lighting

        local shine = Instance.new("BloomEffect")
        shine.Name          = "livyii_gloss_shine"
        shine:SetAttribute("livyii_shader", true)
        shine.Intensity     = 0.6 + 0.6 * k
        shine.Size          = 6 + 5 * k
        shine.Threshold     = 2.0 - 0.4 * k
        shine.Parent        = Lighting

        local rays = Instance.new("SunRaysEffect")
        rays:SetAttribute("livyii_shader", true)
        rays.Intensity    = 0.15 + 0.15 * k
        rays.Spread       = 0.9 + 0.1 * k
        rays.Parent       = Lighting
    end

    local hitboxHighlights = {}
    local function applyHitboxTo(plr)
        local char = plr.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        if Config.HitboxExpander and Config.HitboxActive then
            pcall(function() hrp.Size = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize) end)
            local h = hitboxHighlights[plr]
            if not h or not h.Parent then
                h = Instance.new("Highlight")
                h.Name = "livyii_hitbox"
                h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                h.Adornee = char
                h.Parent = char
                hitboxHighlights[plr] = h
            end

    local function getCursorPos()
        if UserInputService.TouchEnabled and not UserInputService.MouseEnabled then
            local vp = Camera.ViewportSize
            return Vector2.new(vp.X / 2, vp.Y / 2)
        end
        return Vector2.new(Mouse.X, Mouse.Y)
    end
    function GetClosestPlayer(fov)
        local closest, bestDist = nil, math.huge
        local cursor = getCursorPos()
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local sp, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        local d = (Vector2.new(sp.X, sp.Y) - cursor).Magnitude
                        if d <= fov and d < bestDist then bestDist = d; closest = plr end
                    end
                end
            end
        end
        return closest
    end

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

            dragStart = input.Position; dragOrigin = Frame.Position
        end

            local delta = input.Position - dragStart
            Frame.Position = UDim2.new(dragOrigin.X.Scale, dragOrigin.X.Offset + delta.X, dragOrigin.Y.Scale, dragOrigin.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input == dragInput then dragActive = false; dragInput = nil end
    end)

    Submit.Size = UDim2.new(1, -30, 0, 34)

    Submit.BackgroundColor3 = P.BTN_BG_ON
    Submit.Text = "🍮  unlock  🍮"

    corner(Submit, 12)

    local Status = Instance.new("TextLabel")
    Status.Size = UDim2.new(1, -30, 0, 16)
    Status.Position = UDim2.new(0, 15, 1, -22)
    Status.BackgroundTransparency = 1
    Status.Text = ""
    Status.TextColor3 = P.BTN_TEXT
    Status.Font = Enum.Font.GothamMedium
    Status.TextSize = 10
    Status.TextXAlignment = Enum.TextXAlignment.Left
    Status.Parent = Frame

    Submit.MouseButton1Down:Connect(function()
        tween(Submit, { BackgroundColor3 = P.FILL_A }, 0.08)
    end)
    Submit.MouseButton1Up:Connect(function()
        tween(Submit, { BackgroundColor3 = P.BTN_BG_ON }, 0.15, Enum.EasingStyle.Back)
    end)

    local function trySubmit()
        local entered = string.lower(string.gsub(Box.Text, "%s", ""))
        local currentKey = getCurrentKey()

        if entered == "" then
            Status.Text = "type something first! 🍮"
            Status.TextColor3 = P.ERROR
            local orig = Frame.Position
            tween(Frame, { Position = orig + UDim2.new(0, 5, 0, 0) }, 0.05)
            task.wait(0.06)
            tween(Frame, { Position = orig - UDim2.new(0, 5, 0, 0) }, 0.05)
            task.wait(0.06)
            tween(Frame, { Position = orig }, 0.05)
            return
        end

        if entered == currentKey then
            Status.Text = "correct! loading... ✨"
            Status.TextColor3 = P.SUCCESS
            tween(Submit, { BackgroundColor3 = P.SUCCESS }, 0.2)
            task.wait(0.6)
            KeyGui:Destroy()
            startMain()
        else
            Status.Text = "wrong key! try again 💔"
            Status.TextColor3 = P.ERROR
            Box.Text = ""
            local orig = Frame.Position
            for _ = 1, 3 do
                tween(Frame, { Position = orig + UDim2.new(0, 6, 0, 0) }, 0.05)
                task.wait(0.06)
                tween(Frame, { Position = orig - UDim2.new(0, 6, 0, 0) }, 0.05)
                task.wait(0.06)
            end
            tween(Frame, { Position = orig }, 0.08)
        end
    end

    Submit.MouseButton1Click:Connect(trySubmit)
    Box.FocusLost:Connect(function(enterPressed)
        if enterPressed then trySubmit() end
    end)

    Frame.Size = UDim2.new(0, 0, 0, 0)
    tween(Frame, { Size = UDim2.new(0, 260, 0, 180) }, 0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end
