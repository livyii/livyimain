local function tween(o,p,t,s,d) TweenService:Create(o, TweenInfo.new(t or .22, s or Enum.EasingStyle.Quint, d or Enum.EasingDirection.Out), p):Play() end
local function corner(p,r) local c=Instance.new("UICorner",p); c.CornerRadius=UDim.new(0,r or 16); return c end
local function stroke(p,c,t) local s=Instance.new("UIStroke",p); s.Color=c or P.STROKE; s.Thickness=t or 1; s.ApplyStrokeMode=Enum.ApplyStrokeMode.Border; return s end
local function grad(p,a,b,r) local g=Instance.new("UIGradient",p); g.Color=ColorSequence.new(a,b); g.Rotation=r or 90; return g end
local function pop(o,a) local b=o.Size; tween(o,{Size=UDim2.new(b.X.Scale,b.X.Offset+(a or 6),b.Y.Scale,b.Y.Offset+(a or 6))},.1,Enum.EasingStyle.Back); task.wait(.1); tween(o,{Size=b},.18,Enum.EasingStyle.Back) end
local function parentGui(g)
    if gethui then g.Parent=gethui()
    elseif syn and syn.protect_gui then syn.protect_gui(g); g.Parent=CoreGui
    else g.Parent=CoreGui end
end
local function getCursor()
    if Platform.isMobile then local vp=workspace.CurrentCamera.ViewportSize; return Vector2.new(vp.X/2,vp.Y/2) end
    return Vector2.new(Mouse.X,Mouse.Y)
end
local function heartPop(parent,pos,count)
    count=count or 4
    for i=1,count do
        local h=Instance.new("TextLabel")
        h.Size=UDim2.new(0,20,0,20)
        h.Position=UDim2.new(pos.X.Scale,pos.X.Offset-10,pos.Y.Scale,pos.Y.Offset-10)
        h.BackgroundTransparency=1
        h.Text=({"💕","💖","💗","🌸"})[math.random(4)]
        h.TextSize=14; h.Font=Enum.Font.GothamBold; h.TextTransparency=0; h.ZIndex=200; h.Parent=parent
        local a=math.random()*math.pi*2; local d=20+math.random()*16
        tween(h,{Position=UDim2.new(pos.X.Scale,pos.X.Offset-10+math.cos(a)*d,pos.Y.Scale,pos.Y.Offset-10+math.sin(a)*d),TextTransparency=1,Rotation=math.random(-30,30)},.8,Enum.EasingStyle.Quart)
        task.delay(.85,function() h:Destroy() end)
    end
end

    local s=Instance.new("TextLabel"); s.Size=UDim2.new(1,0,0,26); s.Position=UDim2.new(0,0,0,78); s.BackgroundTransparency=1; s.Text="loading"; s.TextColor3=P.HEADER_TEXT; s.Font=Enum.Font.GothamMedium; s.TextSize=14; s.TextTransparency=1; s.Parent=c
    local d=Instance.new("TextLabel"); d.Size=UDim2.new(1,0,0,20); d.Position=UDim2.new(0,0,0,100); d.BackgroundTransparency=1; d.Text=""; d.TextColor3=P.PINK; d.Font=Enum.Font.GothamBold; d.TextSize=14; d.TextTransparency=1; d.Parent=c
    tween(blur,{Size=24},.6,Enum.EasingStyle.Quad,Enum.EasingDirection.Out); tween(veil,{BackgroundTransparency=.35},.6)
    task.wait(.25)
    t.TextSize=20; tween(t,{TextTransparency=0},.6)
    TweenService:Create(t,TweenInfo.new(.9,Enum.EasingStyle.Back),{TextSize=34}):Play()
    task.wait(.4)
    tween(s,{TextTransparency=0},.4); tween(d,{TextTransparency=0},.4)
    task.spawn(function()
        local seq={".","..","...","...."}; local i=1
        while d.Parent do d.Text=seq[i]; i=i+1; if i>#seq then i=1 end; task.wait(.22) end
    end)
    task.wait(1.6)
    s.Text="loaded ✨"; s.TextColor3=P.PINK
    task.wait(.85)
    tween(t,{TextTransparency=1},.5); tween(s,{TextTransparency=1},.5); tween(d,{TextTransparency=1},.5)
    tween(veil,{BackgroundTransparency=1},.6); tween(blur,{Size=0},.6,Enum.EasingStyle.Quad,Enum.EasingDirection.In)
    task.wait(.7); g:Destroy(); blur:Destroy()
end

    -- Get closest player
    local function getClosest(fov)
        local best,bestD=nil,math.huge
        local cur=getCursor()
        local cam=workspace.CurrentCamera
        for _,plr in ipairs(Players:GetPlayers()) do
            if plr~=LP and plr.Character then
                local hrp=plr.Character:FindFirstChild("HumanoidRootPart")
                local hum=plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health>0 then
                    local sp,on=cam:WorldToViewportPoint(hrp.Position)
                    if on then
                        local d=(Vector2.new(sp.X,sp.Y)-cur).Magnitude
                        if d<=fov and d<bestD then bestD=d; best=plr end
                    end
                end
            end
        end
        return best
    end

    -- Hitbox apply
    local function applyHitboxTo(plr)
        local char=plr.Character; if not char then return end
        local hrp=char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
        if Config.HitboxExpander and Config.HitboxActive then
            pcall(function() hrp.Size=Vector3.new(Config.HitboxSize,Config.HitboxSize,Config.HitboxSize) end)
            local h=hitboxHighlights[plr]
            if not h or not h.Parent then
                h=Instance.new("Highlight"); h.Name="livyii_hitbox"
                h.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop; h.Adornee=char; h.Parent=char
                hitboxHighlights[plr]=h
            end
            h.FillColor=Config.HitboxColor; h.FillTransparency=Config.HitboxTransparency
            h.OutlineColor=Config.HitboxColor; h.OutlineTransparency=0
        else
            pcall(function() hrp.Size=Vector3.new(2,2,1) end)
            local h=hitboxHighlights[plr]; if h then pcall(function() h:Destroy() end) end
            hitboxHighlights[plr]=nil
        end
    end
    local function refreshHitboxes()
        for _,p in ipairs(Players:GetPlayers()) do if p~=LP then applyHitboxTo(p) end end
    end

    local function wireFloat(b,onTap)
        local drag,ds,do_,moved,ai=false,nil,nil,false,nil
        b.InputBegan:Connect(function(inp)
            if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
                drag=true; moved=false; ai=inp; ds=inp.Position; do_=b.Position
                tween(b,{Size=UDim2.new(0,S.floatBtn-6,0,S.floatBtn-6)},.1)
            end
        end)
        UIS.InputChanged:Connect(function(inp)
            if not drag or inp~=ai then return end
            if inp.UserInputType==Enum.UserInputType.MouseMovement or inp.UserInputType==Enum.UserInputType.Touch then
                local dx=inp.Position.X-ds.X; local dy=inp.Position.Y-ds.Y
                if math.abs(dx)>6 or math.abs(dy)>6 then moved=true end
                if moved then b.Position=UDim2.new(do_.X.Scale,do_.X.Offset+dx,do_.Y.Scale,do_.Y.Offset+dy) end
            end
        end)
        UIS.InputEnded:Connect(function(inp)
            if not drag or inp~=ai then return end
            if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
                tween(b,{Size=UDim2.new(0,S.floatBtn,0,S.floatBtn)},.2,Enum.EasingStyle.Back)
                if not moved then onTap() end
                drag=false; ai=nil
            end
        end)
    end

    local function toggleVis()
        GuiVisible=not GuiVisible
        local tv=GuiVisible
        if tv then
            Main.Visible=true; Main.BackgroundTransparency=1; Main.Size=UDim2.new(0,S.W-20,0,S.H)
            tween(Main,{BackgroundTransparency=0,Size=UDim2.new(0,S.W,0,S.H)},.28,Enum.EasingStyle.Back)
        else
            tween(Main,{BackgroundTransparency=1,Size=UDim2.new(0,S.W-20,0,S.H)},.2)
            task.wait(.21); Main.Visible=false
        end
        if ToggleBtn then
            local icon=ToggleBtn:FindFirstChild("Icon"); local g=ToggleBtn:FindFirstChildOfClass("UIGradient")
            if icon then icon.Text=tv and "✕" or "💖" end
            if g then g.Color=tv and ColorSequence.new(P.TOGGLE_A,P.TOGGLE_B) or ColorSequence.new(P.HEADER_A,P.HEADER_B) end
            tween(ToggleBtn,{BackgroundColor3=tv and P.TOGGLE_A or P.HEADER_A},.28,Enum.EasingStyle.Back)
        end
    end

                drag=true; moved=false; ai=inp; ds=inp.Position; do_=b.Position

                if bb==b then tween(bb,{BackgroundColor3=P.BTN_BG_ON,TextSize=math.floor(S.tabSize*.55)},.2,Enum.EasingStyle.Back)
                else tween(bb,{BackgroundColor3=Color3.fromRGB(255,253,250),TextSize=math.floor(S.tabSize*.5)},.2) end
            end
            showPage(page); pop(b,4)
        end)
        return b
    end
    local tC=mkTab("🎯",CombatPage)
    local tV=mkTab("🎨",VisualPage)
    local tU=mkTab("🛡️",UtilPage)
    local tCr=mkTab("🏆",CreditsPage)
    tween(tC,{BackgroundColor3=P.BTN_BG_ON,TextSize=math.floor(S.tabSize*.55)},.2,Enum.EasingStyle.Back)

    showPage(MainPage)
    task.delay(2,function()
        showPage(CombatPage)
        for bb in pairs(tabBtns) do
            if bb==tC then tween(bb,{BackgroundColor3=P.BTN_BG_ON,TextSize=math.floor(S.tabSize*.55)},.2,Enum.EasingStyle.Back)
            else tween(bb,{BackgroundColor3=Color3.fromRGB(255,253,250),TextSize=math.floor(S.tabSize*.5)},.2) end
        end
    end)

    MinBtn.MouseButton1Click:Connect(function()
        PageHolder.Visible=not PageHolder.Visible
        Sidebar.Visible=PageHolder.Visible; Content.Visible=PageHolder.Visible; Footer.Visible=PageHolder.Visible
        Main:TweenSize(Main.Size.Y.Offset==S.H and UDim2.new(0,S.W,0,S.headerH) or UDim2.new(0,S.W,0,S.H),Enum.EasingDirection.Out,Enum.EasingStyle.Back,.32,true)
    end)
    CloseBtn.MouseButton1Click:Connect(function() if GuiVisible then toggleVis() end end)

    createToggle()

    -- ═══ LOOPS ═══
    RunService.RenderStepped:Connect(function(dt)
        if not Config.Camlock or not LockedTarget then return end
        local char=LockedTarget.Character
        if not char or not char.Parent then LockedTarget=nil; updateLockVis(); return end
        local hum=char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health<=0 then LockedTarget=nil; updateLockVis(); return end
        local part=char:FindFirstChild(partMap[Config.CamlockPart])
        if not part then return end
        local cam=workspace.CurrentCamera
        if not cam then return end
        local la=part.Position+Vector3.new(0,Config.CamlockYOffset,0)
        local goal=CFrame.new(cam.CFrame.Position,la)
        local alpha=1-math.pow(1-Config.CamlockSmoothness,dt*60)
        cam.CFrame=cam.CFrame:Lerp(goal,alpha)
    end)

    RunService.RenderStepped:Connect(function()
        if not Config.AimAssist then aimHL.Enabled=false; aimBB.Enabled=false; aimFOV.Visible=false; return end
        local cur=getCursor()
        aimFOV.Visible=true
        aimFOV.Size=UDim2.new(0,Config.AimFOV*2,0,Config.AimFOV*2)
        aimFOV.Position=UDim2.new(0,cur.X,0,cur.Y)
        local tgt=getClosest(Config.AimFOV)
        if tgt and tgt.Character then
            local head=tgt.Character:FindFirstChild("Head") or tgt.Character:FindFirstChild("HumanoidRootPart")
            if head then
                aimHL.Adornee=tgt.Character; aimHL.FillColor=Config.AimColor; aimHL.OutlineColor=Config.AimColor; aimHL.Enabled=true
                aimBB.Adornee=head; aimBB.Enabled=true
                aimTagLbl.Text="💖 "..(tgt.DisplayName or tgt.Name)
            end
        else
            aimHL.Enabled=false; aimBB.Enabled=false
        end
    end)

    Players.PlayerAdded:Connect(function(p)
        p.CharacterAdded:Connect(function()
            if Config.HitboxExpander and Config.HitboxActive then task.wait(1); applyHitboxTo(p) end
        end)
    end)

    RunService.Heartbeat:Connect(function()
        if not Config.Antilock then return end
        local char=LP.Character
        if char then
            local hrp=char:FindFirstChild("HumanoidRootPart")
            if hrp and hrp.AssemblyLinearVelocity.Magnitude<1 then hrp.CFrame=hrp.CFrame+Vector3.new(0,.1,0) end
        end
    end)

    print("🍮 livyii.main loaded 💖✨")
end

-- Key system
local function showKey(onSuccess)
    local kg=Instance.new("ScreenGui"); kg.Name="livyii.key"; kg.ResetOnSpawn=false; kg.IgnoreGuiInset=true; kg.DisplayOrder=500; parentGui(kg)
    local f=Instance.new("Frame"); f.Size=UDim2.new(0,270,0,195); f.Position=UDim2.new(.5,-135,.5,-97)
    f.BackgroundColor3=P.BG; f.BorderSizePixel=0; f.Active=true; f.Parent=kg
    corner(f,24); stroke(f,P.PINK,2); grad(f,Color3.fromRGB(255,253,245),Color3.fromRGB(255,242,235),90)
    local h=Instance.new("TextLabel"); h.Size=UDim2.new(1,0,0,40); h.BackgroundColor3=P.HEADER_A; h.BorderSizePixel=0
    h.Text="🍮  livyii.main  🍮"; h.TextColor3=P.HEADER_TEXT; h.Font=Enum.Font.GothamBold; h.TextSize=13; h.Parent=f
    corner(h,24); grad(h,P.HEADER_A,P.HEADER_B,90)
    local fix=Instance.new("Frame"); fix.Size=UDim2.new(1,0,0,24); fix.Position=UDim2.new(0,0,1,-24); fix.BackgroundColor3=P.HEADER_B; fix.BorderSizePixel=0; fix.Parent=h
    local cb=Instance.new("TextButton"); cb.Size=UDim2.new(0,28,0,28); cb.Position=UDim2.new(1,-34,0,6)
    cb.BackgroundColor3=P.BTN_BG; cb.Text="✕"; cb.TextColor3=P.HEADER_TEXT; cb.Font=Enum.Font.GothamBold
    cb.TextSize=13; cb.AutoButtonColor=false; cb.ZIndex=10; cb.Parent=h; corner(cb,14); stroke(cb,P.PINK_SOFT,1.5)
    cb.MouseButton1Click:Connect(function() kg:Destroy(); warn("🍮 livyii.main: key not entered.") end)
    local sub=Instance.new("TextLabel"); sub.Size=UDim2.new(1,-30,0,20); sub.Position=UDim2.new(0,15,0,50)
    sub.BackgroundTransparency=1; sub.Text="💖 enter this week's key 💖"; sub.TextColor3=P.PINK
    sub.Font=Enum.Font.GothamBold; sub.TextSize=11; sub.TextXAlignment=Enum.TextXAlignment.Center; sub.Parent=f
    local box=Instance.new("TextBox"); box.Size=UDim2.new(1,-30,0,36); box.Position=UDim2.new(0,15,0,74)
    box.BackgroundColor3=P.BTN_BG; box.Text=""; box.PlaceholderText="type key here..."
    box.PlaceholderColor3=Color3.fromRGB(230,195,195); box.TextColor3=P.BTN_TEXT_ON
    box.Font=Enum.Font.GothamBold; box.TextSize=13; box.TextXAlignment=Enum.TextXAlignment.Center
    box.ClearTextOnFocus=false; box.Parent=f; corner(box,14); stroke(box,P.PINK_SOFT,1.5)
    local submit=Instance.new("TextButton"); submit.Size=UDim2.new(1,-30,0,36); submit.Position=UDim2.new(0,15,0,118)
    submit.BackgroundColor3=P.BTN_BG_ON; submit.Text="🍮  unlock  🍮"; submit.TextColor3=P.BTN_TEXT_ON
    submit.Font=Enum.Font.GothamBold; submit.TextSize=13; submit.AutoButtonColor=false; submit.Parent=f
    corner(submit,14); stroke(submit,P.PINK,2)
    local status=Instance.new("TextLabel"); status.Size=UDim2.new(1,-30,0,18); status.Position=UDim2.new(0,15,1,-24)
    status.BackgroundTransparency=1; status.Text=""; status.TextColor3=P.BTN_TEXT
    status.Font=Enum.Font.GothamMedium; status.TextSize=10; status.TextXAlignment=Enum.TextXAlignment.Center; status.Parent=f
    local function trySubmit()
        local ent=string.lower(string.gsub(box.Text,"%s",""))
        local cur=getKey()
        if ent=="" then
            status.Text="type something first! 🍮"; status.TextColor3=P.ERROR
            local o=f.Position
            tween(f,{Position=o+UDim2.new(0,5,0,0)},.05); task.wait(.06)
            tween(f,{Position=o-UDim2.new(0,5,0,0)},.05); task.wait(.06)
            tween(f,{Position=o},.05); return
        end
        if ent==cur then
            status.Text="correct! next... 💖"; status.TextColor3=P.SUCCESS
            tween(submit,{BackgroundColor3=P.SUCCESS},.2)
            task.wait(.6); _keyPassed=true; kg:Destroy(); onSuccess()
        else
            status.Text="wrong key! try again 💔"; status.TextColor3=P.ERROR
            box.Text=""
            local o=f.Position
            for _=1,3 do
                tween(f,{Position=o+UDim2.new(0,6,0,0)},.05); task.wait(.06)
                tween(f,{Position=o-UDim2.new(0,6,0,0)},.05); task.wait(.06)
            end
            tween(f,{Position=o},.08)
        end
    end
    submit.MouseButton1Click:Connect(trySubmit)
    box.FocusLost:Connect(function(ep) if ep then trySubmit() end end)
    f.Size=UDim2.new(0,0,0,0); tween(f,{Size=UDim2.new(0,270,0,195)},.4,Enum.EasingStyle.Back)
end
