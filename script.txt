local function tween(obj, props, t, style, dir)
    TweenService:Create(obj,
        TweenInfo.new(t or 0.22, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props):Play()
end
local function corner(parent, r)
    local c = Instance.new("UICorner", parent)
    c.CornerRadius = UDim.new(0, r or 14)
    return c
end
local function stroke(parent, color, thickness)
    local s = Instance.new("UIStroke", parent)
    s.Color = color or P.STROKE
    s.Thickness = thickness or 1
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return s
end
local function gradient(parent, a, b, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new(a, b)
    g.Rotation = rot or 90
    return g
end
local function pop(obj, amount)
    local base = obj.Size
    tween(obj, { Size = UDim2.new(base.X.Scale, base.X.Offset + (amount or 6), base.Y.Scale, base.Y.Offset + (amount or 6)) }, 0.08, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    task.wait(0.09)
    tween(obj, { Size = base }, 0.15, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end

local function getCursorPos()
    if Platform.isMobile then
        local vp = Camera.ViewportSize
        return Vector2.new(vp.X / 2, vp.Y / 2)
    end
    return Vector2.new(Mouse.X, Mouse.Y)
end

    if gethui then gui.Parent = gethui()

    else gui.Parent = CoreGui end
end

    Main.Active = false; Main.Draggable = false; Main.Parent = ScreenGui
    corner(Main, 24); stroke(Main, P.STROKE_SOFT, 1.5)

    do
        local sh = Instance.new("Frame")
        sh.Size = UDim2.new(1, 0, 1, 0); sh.Position = UDim2.new(0, 0, 0, 5)
        sh.BackgroundColor3 = P.SHADOW; sh.BackgroundTransparency = 0.7

        corner(sh, 24)
    end

    do
        local g = Instance.new("UIGradient", Main)
        g.Color = ColorSequence.new(Color3.fromRGB(255, 253, 240), Color3.fromRGB(255, 248, 220))
        g.Rotation = 90
    end

    local Title = Instance.new("TextLabel")
    Title.Size = UDim2.new(1, 0, 0, s.headerH)
    Title.BackgroundColor3 = P.HEADER_A; Title.BorderSizePixel = 0
    Title.Text = "🍮  livyii.main  🍮"
    Title.TextColor3 = P.HEADER_TEXT; Title.Font = Enum.Font.GothamBold
    Title.TextSize = s.titleSize; Title.TextYAlignment = Enum.TextYAlignment.Center
    Title.Parent = Main; corner(Title, 24)
    gradient(Title, P.HEADER_A, P.HEADER_B, 90)

    local TitleFix = Instance.new("Frame")
    TitleFix.Size = UDim2.new(1, 0, 0, 24); TitleFix.Position = UDim2.new(0, 0, 1, -24)
    TitleFix.BackgroundColor3 = P.HEADER_B; TitleFix.BorderSizePixel = 0; TitleFix.Parent = Title

    do
        local hl = Instance.new("Frame")
        hl.Size = UDim2.new(1, -30, 0, 1); hl.Position = UDim2.new(0, 15, 0, 2)
        hl.BackgroundColor3 = Color3.fromRGB(255, 255, 250); hl.BackgroundTransparency = 0.4
        hl.BorderSizePixel = 0; hl.ZIndex = 3; hl.Parent = Title
        corner(hl, 1)
    end

    if profileName and profileData then
        local badge = Instance.new("TextLabel")
        badge.Size = UDim2.new(0, 100, 0, 14); badge.Position = UDim2.new(0, 8, 1, -18)
        badge.BackgroundColor3 = profileData.color; badge.BackgroundTransparency = 0.35
        badge.Text = profileData.emoji .. " " .. profileData.role
        badge.TextColor3 = P.HEADER_TEXT; badge.Font = Enum.Font.GothamBold
        badge.TextSize = 8; badge.ZIndex = 5; badge.Parent = Title
        corner(badge, 7)
    else
        local platBadge = Instance.new("TextLabel")
        platBadge.Size = UDim2.new(0, 70, 0, 14); platBadge.Position = UDim2.new(0, 8, 1, -18)
        platBadge.BackgroundColor3 = Platform.isMobile and Color3.fromRGB(255, 200, 220) or Color3.fromRGB(200, 220, 255)
        platBadge.BackgroundTransparency = 0.4
        platBadge.Text = Platform.isMobile and "📱 mobile" or "🖥️ pc"
        platBadge.TextColor3 = P.HEADER_TEXT; platBadge.Font = Enum.Font.GothamBold
        platBadge.TextSize = 8; platBadge.ZIndex = 5; platBadge.Parent = Title
        corner(platBadge, 7)
    end

    local DragZone = Instance.new("Frame")
    DragZone.Size = UDim2.new(1, -80, 1, 0)
    DragZone.BackgroundTransparency = 1; DragZone.Active = true; DragZone.Parent = Title

    local dragging, dragStart, dragOrigin = false, nil, nil
    local dragActiveInput = nil
    DragZone.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragActiveInput = input
            dragStart = input.Position; dragOrigin = Main.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging or input ~= dragActiveInput then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(dragOrigin.X.Scale, dragOrigin.X.Offset + delta.X, dragOrigin.Y.Scale, dragOrigin.Y.Offset + delta.Y)
        end
    end)

    local function makeSlider(parent, label, minV, maxV, default, suffix, decimals, onChange)
        local sliderH = Platform.isMobile and 48 or 42
        local Holder = Instance.new("Frame")
        Holder.Size = UDim2.new(1, 0, 0, sliderH)

        local TrackHit = Instance.new("TextButton")
        TrackHit.Size = UDim2.new(1, -4, 0, trackH)
        TrackHit.Position = UDim2.new(0, 2, 0, Platform.isMobile and 22 or 20)
        TrackHit.BackgroundTransparency = 1; TrackHit.Text = ""
        TrackHit.AutoButtonColor = false; TrackHit.Active = true
        TrackHit.ZIndex = 5; TrackHit.Parent = Holder

        local Track = Instance.new("Frame")
        Track.Size = UDim2.new(1, -4, 0, Platform.isMobile and 12 or 10)
        Track.Position = UDim2.new(0, 2, 0.5, -5)
        Track.BackgroundColor3 = P.TRACK; Track.BorderSizePixel = 0
        Track.ZIndex = 2; Track.Parent = TrackHit; corner(Track, 6)
        local ts = Instance.new("UIStroke", Track)
        ts.Color = P.STROKE_SOFT; ts.Thickness = 1; ts.Transparency = 0.4

        local Fill = Instance.new("Frame")
        Fill.Size = UDim2.new(0, 0, 1, 0); Fill.BackgroundColor3 = P.FILL_A
        Fill.BorderSizePixel = 0; Fill.ZIndex = 3; Fill.Parent = Track
        corner(Fill, 6); gradient(Fill, P.FILL_A, P.FILL_B, 0)

        local knobSize = Platform.isMobile and 22 or 18
        local Knob = Instance.new("Frame")
        Knob.Size = UDim2.new(0, knobSize, 0, knobSize)
        Knob.AnchorPoint = Vector2.new(0.5, 0.5)
        Knob.Position = UDim2.new(0, 0, 0.5, 0)
        Knob.BackgroundColor3 = Color3.fromRGB(255, 255, 252)
        Knob.ZIndex = 4; Knob.Parent = Track; corner(Knob, knobSize / 2)
        stroke(Knob, P.FILL_B, 2)

        local function fmt(v)
            if decimals == 0 then return tostring(math.floor(v + 0.5)) .. suffix end
            return string.format("%." .. tostring(decimals) .. "f", v) .. suffix
        end
        local function apply(v)
            local pct = (v - minV) / (maxV - minV)
            tween(Fill, { Size = UDim2.new(pct, 0, 1, 0) }, 0.08)
            tween(Knob, { Position = UDim2.new(pct, 0, 0.5, 0) }, 0.08)
            Val.Text = fmt(v); onChange(v)
        end
        Val.Text = fmt(default)

        local draggingS = false
        local activeSliderInput = nil
        local function fromX(x)
            local rel = math.clamp((x - Track.AbsolutePosition.X) / math.max(Track.AbsoluteSize.X, 1), 0, 1)
            local v = minV + (maxV - minV) * rel
            if decimals == 0 then v = math.floor(v + 0.5) end
            apply(v)
        end
        TrackHit.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                draggingS = true; activeSliderInput = input
                fromX(input.Position.X)
                tween(Knob, { Size = UDim2.new(0, knobSize + 4, 0, knobSize + 4) }, 0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not draggingS or input ~= activeSliderInput then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                fromX(input.Position.X)
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if not draggingS or input ~= activeSliderInput then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(Knob, { Size = UDim2.new(0, knobSize, 0, knobSize) }, 0.15, Enum.EasingStyle.Back)
                draggingS = false; activeSliderInput = nil
            end
        end)
        task.defer(function() apply(default) end)
        return Holder
    end

        local presets = {
            Color3.fromRGB(255, 220, 130), Color3.fromRGB(255, 200, 130),
            Color3.fromRGB(255, 180, 100), Color3.fromRGB(220, 160, 90),
            Color3.fromRGB(255,  60,  60), Color3.fromRGB(255, 150,  80),

            return string.format("#%02X%02X%02X",
                math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
        end
        local function rebuildBrightGrad()
            local base = Color3.fromHSV(currentHue, 1, 1)
            brightGrad.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0.0, Color3.new(1, 1, 1)),
                ColorSequenceKeypoint.new(0.5, base),
                ColorSequenceKeypoint.new(1.0, Color3.new(0, 0, 0)),
            })
        end

            Preview.BackgroundColor3 = c
            HexLbl.Text = toHex(c)

            local rel = math.clamp((x - HueStrip.AbsolutePosition.X) / math.max(HueStrip.AbsoluteSize.X, 1), 0, 1)
            currentHue = rel * 0.9999
            HueCursor.Position = UDim2.new(rel, 0, 0.5, 0)

            local rel = math.clamp((x - BrightStrip.AbsolutePosition.X) / math.max(BrightStrip.AbsoluteSize.X, 1), 0, 1)

            if brightDrag and input == brightInput then brightFromX(input.Position.X) end
        end)

    local function applyShaders()
        clearShaders()
        if not Config.ShaderEnabled then return end
        local preset = ShaderPresets[Config.ShaderPreset] or ShaderPresets.pudding
        local k = Config.ShaderIntensity
        local perfMul = Platform.isMobile and 0.75 or 1

        local atmo = Instance.new("Atmosphere")
        atmo:SetAttribute("livyii_shader", true)
        atmo.Density = (0.06 + 0.08 * k) * perfMul
        atmo.Offset  = 0.08
        atmo.Color   = preset.atmo
        atmo.Decay   = preset.decay
        atmo.Glare   = (0.20 + 0.25 * k) * perfMul
        atmo.Haze    = (0.04 + 0.06 * k) * perfMul
        atmo.Parent  = Lighting

        local cc = Instance.new("ColorCorrectionEffect")
        cc:SetAttribute("livyii_shader", true)
        cc.Brightness = 0.02 + 0.04 * k
        cc.Contrast   = 0.15 + 0.12 * k
        cc.Saturation = 0.25 + 0.20 * k
        cc.TintColor  = preset.tint
        cc.Parent     = Lighting

        local bloom = Instance.new("BloomEffect")
        bloom:SetAttribute("livyii_shader", true)
        bloom.Intensity = (0.60 + 0.50 * k) * perfMul
        bloom.Size      = 6 + 5 * k
        bloom.Threshold = 0.95 - 0.10 * k
        bloom.Parent    = Lighting

        local shine = Instance.new("BloomEffect")
        shine.Name          = "livyii_gloss_shine"
        shine:SetAttribute("livyii_shader", true)
        shine.Intensity     = (0.35 + 0.35 * k) * perfMul
        shine.Size          = 3 + 2 * k
        shine.Threshold     = 1.6 - 0.20 * k
        shine.Parent        = Lighting

        local rays = Instance.new("SunRaysEffect")
        rays:SetAttribute("livyii_shader", true)
        rays.Intensity = (0.08 + 0.10 * k) * perfMul
        rays.Spread    = 0.9
        rays.Parent    = Lighting
    end

    local hitboxHighlights = {}
    local function applyHitboxTo(plr)
        local char = plr.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        if Config.HitboxExpander and Config.HitboxActive then
            pcall(function() hrp.Size = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize) end)
            local h = hitboxHighlights[plr]
            if not h or not h.Parent then
                h = Instance.new("Highlight")
                h.Name = "livyii_hitbox"
                h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                h.Adornee = char; h.Parent = char
                hitboxHighlights[plr] = h
            end
            h.FillColor = Config.HitboxColor
            h.FillTransparency = Config.HitboxTransparency
            h.OutlineColor = Config.HitboxColor
            h.OutlineTransparency = 0
        else
            pcall(function() hrp.Size = Vector3.new(2, 2, 1) end)
            local h = hitboxHighlights[plr]
            if h then pcall(function() h:Destroy() end) end
            hitboxHighlights[plr] = nil
        end
    end

    function GetClosestPlayer(fov)
        local closest, bestDist = nil, math.huge
        local cursor = getCursorPos()
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local sp, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        local d = (Vector2.new(sp.X, sp.Y) - cursor).Magnitude
                        if d <= fov and d < bestDist then bestDist = d; closest = plr end
                    end
                end
            end
        end
        return closest
    end

    local function wireFloatBtn(b, onTap)
        local draggingB, dragStartB, startPosB, moved, activeInput = false, nil, nil, false, nil
        b.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                draggingB = true; moved = false; activeInput = input
                dragStartB = input.Position; startPosB = b.Position
                tween(b, { Size = UDim2.new(0, s.floatBtn - 6, 0, s.floatBtn - 6) }, 0.1)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not draggingB or input ~= activeInput then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                local dx = input.Position.X - dragStartB.X
                local dy = input.Position.Y - dragStartB.Y
                if math.abs(dx) > 6 or math.abs(dy) > 6 then moved = true end
                if moved then
                    b.Position = UDim2.new(startPosB.X.Scale, startPosB.X.Offset + dx, startPosB.Y.Scale, startPosB.Y.Offset + dy)
                end
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if not draggingB or input ~= activeInput then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                tween(b, { Size = UDim2.new(0, s.floatBtn, 0, s.floatBtn) }, 0.18, Enum.EasingStyle.Back)
                if not moved then onTap() end
                draggingB = false; activeInput = nil
            end
        end)
    end

local function showKeySystem(onSuccess)
    local KeyGui = Instance.new("ScreenGui")

    KeyGui.IgnoreGuiInset = true; KeyGui.DisplayOrder = 500
    KeyGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    parentGui(KeyGui)

    Frame.Position = UDim2.new(0.5, -130, 0.5, -90)
    Frame.BackgroundColor3 = P.BG; Frame.BorderSizePixel = 0

    corner(Frame, 22); stroke(Frame, P.STROKE_SOFT, 1.5)

    do
        local sh = Instance.new("Frame")
        sh.Size = UDim2.new(1, 0, 1, 0); sh.Position = UDim2.new(0, 0, 0, 5)
        sh.BackgroundColor3 = P.SHADOW; sh.BackgroundTransparency = 0.7
        sh.BorderSizePixel = 0; sh.ZIndex = -1; sh.Parent = Frame
        corner(sh, 22)
    end

    do
        local g = Instance.new("UIGradient", Frame)
        g.Color = ColorSequence.new(Color3.fromRGB(255, 253, 240), Color3.fromRGB(255, 245, 220))
        g.Rotation = 90
    end

    local Header = Instance.new("TextLabel")
    Header.Size = UDim2.new(1, 0, 0, 36); Header.BackgroundColor3 = P.HEADER_A
    Header.BorderSizePixel = 0; Header.Text = "🍮  livyii.main  🍮"
    Header.TextColor3 = P.HEADER_TEXT; Header.Font = Enum.Font.GothamBold
    Header.TextSize = 12; Header.Parent = Frame
    corner(Header, 22); gradient(Header, P.HEADER_A, P.HEADER_B, 90)

    do
        local fix = Instance.new("Frame")
        fix.Size = UDim2.new(1, 0, 0, 22); fix.Position = UDim2.new(0, 0, 1, -22)
        fix.BackgroundColor3 = P.HEADER_B; fix.BorderSizePixel = 0; fix.Parent = Header
    end

    local dragActive, dragStart, dragOrigin, dragInput = false, nil, nil, nil
    Header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragActive = true; dragInput = input
            dragStart = input.Position; dragOrigin = Frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragActive or input ~= dragInput then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            Frame.Position = UDim2.new(dragOrigin.X.Scale, dragOrigin.X.Offset + delta.X, dragOrigin.Y.Scale, dragOrigin.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input == dragInput then dragActive = false; dragInput = nil end
    end)

    local CloseB = Instance.new("TextButton")
    CloseB.Size = UDim2.new(0, 26, 0, 26); CloseB.Position = UDim2.new(1, -32, 0, 5)
    CloseB.BackgroundColor3 = P.BTN_BG; CloseB.Text = "✕"
    CloseB.TextColor3 = P.HEADER_TEXT; CloseB.Font = Enum.Font.GothamBold
    CloseB.TextSize = 12; CloseB.AutoButtonColor = false
    CloseB.ZIndex = 10; CloseB.Parent = Header

    Box.Size = UDim2.new(1, -30, 0, 34); Box.Position = UDim2.new(0, 15, 0, 70)
    Box.BackgroundColor3 = P.BTN_BG; Box.Text = ""
    Box.PlaceholderText = "type key here..."
    Box.PlaceholderColor3 = Color3.fromRGB(220, 205, 175)
    Box.TextColor3 = P.BTN_TEXT_ON; Box.Font = Enum.Font.GothamMedium
    Box.TextSize = 12; Box.TextXAlignment = Enum.TextXAlignment.Center
    Box.ClearTextOnFocus = false; Box.Parent = Frame
    corner(Box, 13); stroke(Box, P.STROKE_SOFT, 1)

    local Submit = Instance.new("TextButton")
    Submit.Size = UDim2.new(1, -30, 0, 34); Submit.Position = UDim2.new(0, 15, 0, 112)
    Submit.BackgroundColor3 = P.BTN_BG_ON; Submit.Text = "🍮  unlock  🍮"
    Submit.TextColor3 = P.BTN_TEXT_ON; Submit.Font = Enum.Font.GothamBold
    Submit.TextSize = 12; Submit.AutoButtonColor = false; Submit.Parent = Frame
    corner(Submit, 13); stroke(Submit, P.STROKE_SOFT, 1)

    local Status = Instance.new("TextLabel")
    Status.Size = UDim2.new(1, -30, 0, 16); Status.Position = UDim2.new(0, 15, 1, -22)
    Status.BackgroundTransparency = 1; Status.Text = ""
    Status.TextColor3 = P.BTN_TEXT; Status.Font = Enum.Font.GothamMedium
    Status.TextSize = 10; Status.TextXAlignment = Enum.TextXAlignment.Left; Status.Parent = Frame

    Submit.MouseButton1Down:Connect(function() tween(Submit, { BackgroundColor3 = P.FILL_A }, 0.08) end)
    Submit.MouseButton1Up:Connect(function() tween(Submit, { BackgroundColor3 = P.BTN_BG_ON }, 0.15, Enum.EasingStyle.Back) end)

    local function trySubmit()
        local entered = string.lower(string.gsub(Box.Text, "%s", ""))
        local currentKey = getCurrentKey()

        if entered == "" then
            Status.Text = "type something first! 🍮"
            Status.TextColor3 = P.ERROR
            local orig = Frame.Position
            tween(Frame, { Position = orig + UDim2.new(0, 5, 0, 0) }, 0.05); task.wait(0.06)
            tween(Frame, { Position = orig - UDim2.new(0, 5, 0, 0) }, 0.05); task.wait(0.06)
            tween(Frame, { Position = orig }, 0.05)
            return
        end

        if entered == currentKey then
            Status.Text = "correct! next... ✨"
            Status.TextColor3 = P.SUCCESS
            tween(Submit, { BackgroundColor3 = P.SUCCESS }, 0.2)
            task.wait(0.6)
            _keyPassed = true          -- 🔓 unlock the gate
            KeyGui:Destroy()
            onSuccess()
        else
            Status.Text = "wrong key! try again 💔"
            Status.TextColor3 = P.ERROR
            Box.Text = ""
            local orig = Frame.Position
            for _ = 1, 3 do
                tween(Frame, { Position = orig + UDim2.new(0, 6, 0, 0) }, 0.05); task.wait(0.06)
                tween(Frame, { Position = orig - UDim2.new(0, 6, 0, 0) }, 0.05); task.wait(0.06)
            end
            tween(Frame, { Position = orig }, 0.08)
        end
    end

    Submit.MouseButton1Click:Connect(trySubmit)
    Box.FocusLost:Connect(function(enterPressed) if enterPressed then trySubmit() end end)

    Frame.Size = UDim2.new(0, 0, 0, 0)
    tween(Frame, { Size = UDim2.new(0, 260, 0, 180) }, 0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end
