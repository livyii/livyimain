-- ===================== SILENT AIM (from hcsilente) =====================
local silentAimCachedPart = nil

local function isHoldingRevolver()
    if not _G.RevolverBypass then return false end
    local char = Player.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        local n = string.lower(tool.Name)
        if string.find(n, "revolver") or string.find(n, "rev") then return true end
    end
    return false
end

local function getHum(p) local c = p and p.Character; return c and c:FindFirstChildOfClass("Humanoid") end
local function getHRP(p) local c = p and p.Character; return c and c:FindFirstChild("HumanoidRootPart") end
local function isAlive(p) local h = getHum(p); return h and h.Health > 0 end

local function wallBetween(pos)
    if not _G.WallCheck then return false end
    local ro = cam.CFrame.Position
    local rd = pos - ro
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {Player.Character}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local hit = workspace:Raycast(ro, rd, params)
    if not hit then return false end
    for _, p in pairs(Players:GetPlayers()) do
        if p.Character and hit.Instance:IsDescendantOf(p.Character) then return false end
    end
    return true
end

local function safeWorldToViewportPoint(pos)
    local sp, on = Vector3.new(), false
    pcall(function() sp, on = cam:WorldToViewportPoint(pos) end)
    return sp, on
end

local function getClosestBodyPart(char)
    local closestPart, shortestDist = nil, math.huge
    local mousePos = UserInputService:GetMouseLocation()
    for _, pName in ipairs(bodyPartsList) do
        local part = char:FindFirstChild(pName)
        if part then
            local sp, on = safeWorldToViewportPoint(part.Position)
            local dist = on and (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude or math.huge
            if dist < shortestDist then shortestDist = dist; closestPart = part end
        end
    end
    return closestPart or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end

local function getTargetPart(char)
    if not char then return nil end
    if _G.SilentAimClosestPart then return getClosestBodyPart(char) end
    return char:FindFirstChild(_G.SilentAimPart) or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end

local function getClosestPlayerToCursor()
    if isHoldingRevolver() then return nil end
    local closestPart, shortestDist = nil, _G.FOV_RADIUS
    local mousePos = UserInputService:GetMouseLocation()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= Player and isAlive(p) then
            if _G.Whitelist and _G.Whitelist[p.UserId] then continue end
            local hrp = getHRP(p)
            if hrp then
                local dist3D = (hrp.Position - cam.CFrame.Position).Magnitude
                if dist3D <= _G.SilentAimMaxDist then
                    local sp, on = safeWorldToViewportPoint(hrp.Position)
                    if on then
                        local dist2D = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
                        if dist2D < shortestDist then
                            local part = getTargetPart(p.Character)
                            if part and not wallBetween(part.Position) then
                                shortestDist = dist2D
                                closestPart = part
                            end
                        end
                    end
                end
            end
        end
    end
    return closestPart
end

-- Metatable silent aim
pcall(function()
    local _grm = getrawmetatable(game)
    local _oldIndex = _grm.__index
    setreadonly(_grm, false)
    _grm.__index = function(self, key)
        if not checkcaller() and self == mouse and _G.SilentAimEnabled and not isHoldingRevolver() then
            if (key == "Hit" or key == "Target" or key == "UnitRay") and silentAimCachedPart then
                if math.random(1, 100) <= (_G.SilentAimHitChance or 100) then
                    local origin = cam.CFrame.Position
                    local hitPos = silentAimCachedPart.Position + Vector3.new(
                        silentAimCachedPart.AssemblyLinearVelocity.X * (_G.SilentAimPredX or 0),
                        silentAimCachedPart.AssemblyLinearVelocity.Y * (_G.SilentAimPredY or 0),
                        silentAimCachedPart.AssemblyLinearVelocity.Z * (_G.SilentAimPredX or 0)
                    )
                    if key == "UnitRay" then return Ray.new(origin, (hitPos - origin).Unit)
                    elseif key == "Hit" then return CFrame.new(hitPos)
                    elseif key == "Target" then return silentAimCachedPart end
                end
            end
        end
        return _oldIndex(self, key)
    end
    setreadonly(_grm, true)
end)

-- ===================== CAMLOCK LOGIC =====================
local function isKnocked(character)
    local bodyEffects = character:FindFirstChild("BodyEffects")
    if bodyEffects and bodyEffects:FindFirstChild("K.O") then return bodyEffects["K.O"].Value end
    return false
end

local function IsGrabbed(player)
    return player and player.Character and player.Character:FindFirstChild("GRABBING_CONSTRAINT") ~= nil
end

local function getClosestPartToMouse(char)
    local m = UserInputService:GetMouseLocation()
    local nearestPart, nearestDist = nil, math.huge
    for _, name in ipairs(bodyPartsList) do
        local part = char:FindFirstChild(name)
        if part then
            local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - Vector2.new(m.X, m.Y)).Magnitude
                if dist < nearestDist then nearestDist = dist nearestPart = part end
            end
        end
    end
    return nearestPart
end

local function GetCamlockHitPosition(Target)
    if not Target or not Target.Character then return nil end
    local Character = Target.Character
    local NearestPart = getClosestPartToMouse(Character)
    if not NearestPart then return nil end
    local HitPosition
    if _G.CamlockHitPart == "Closest Point" or _G.CamlockHitPart == "Closest Part" then
        HitPosition = NearestPart.Position
    else
        local part = Character:FindFirstChild(_G.CamlockHitPart)
        HitPosition = part and part.Position
    end
    if not HitPosition then return nil end
    if _G.CamlockPredictionEnabled then
        local RootPart = Character:FindFirstChild("HumanoidRootPart")
        if RootPart then
            HitPosition = HitPosition + RootPart.Velocity * Vector3.new(_G.CamlockPredictionX, _G.CamlockPredictionY, _G.CamlockPredictionZ)
        end
    end
    return HitPosition
end

local function GetBestCamlockTarget()
    local Closest, Distance = nil, _G.CamlockFOVRadius > 0 and _G.CamlockFOVRadius or math.huge
    local MousePosition = UserInputService:GetMouseLocation()
    for _, targetPlayer in ipairs(Players:GetPlayers()) do
        if targetPlayer == Player then continue end
        if not targetPlayer.Character then continue end
        local Character = targetPlayer.Character
        local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
        if not HumanoidRootPart then continue end
        local Position, OnScreen = cam:WorldToViewportPoint(HumanoidRootPart.Position)
        if not OnScreen then continue end
        if _G.CamlockConditionsForceField and Character:FindFirstChild("Forcefield") then continue end
        if _G.CamlockConditionsKnocked and isKnocked(Character) then continue end
        if _G.CamlockConditionsSelfKnocked and isKnocked(Player.Character) then continue end
        if _G.CamlockConditionsCarried and IsGrabbed(targetPlayer) then continue end
        local Magnitude = (Vector2.new(Position.X, Position.Y) - MousePosition).Magnitude
        if Magnitude < Distance then Closest = targetPlayer Distance = Magnitude end
    end
    return Closest
end

local Camlock = { Target = nil, Active = false, Connection = nil }

local function IsHoldingGun()
    local char = Player.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return false end
    if tool:FindFirstChild("Ammo") or tool:FindFirstChild("Magazine") then return true end
    return false
end

local function UpdateCamlock()
    if not _G.CamlockEnabled then
        Camlock.Active = false
        Camlock.Target = nil
        return
    end
    if _G.CamlockAutoToggle and isPC then
        if not IsHoldingGun() then
            Camlock.Active = false
            Camlock.Target = nil
            return
        end
        if not Camlock.Active or not Camlock.Target or not Camlock.Target.Character then
            local target = GetBestCamlockTarget()
            if target then Camlock.Target = target Camlock.Active = true
            else Camlock.Active = false Camlock.Target = nil end
            return
        end
    else
        if not Camlock.Active then return end
    end
    if not Camlock.Active then return end
    if not Camlock.Target or not Camlock.Target.Character then Camlock.Active = false return end
    local Character = Camlock.Target.Character
    if not Character:FindFirstChild("HumanoidRootPart") then Camlock.Active = false return end
    if _G.CamlockConditionsForceField and Character:FindFirstChild("Forcefield") then return end
    if _G.CamlockConditionsKnocked and isKnocked(Character) then return end
    if _G.CamlockConditionsSelfKnocked and isKnocked(Player.Character) then return end
    if _G.CamlockConditionsCarried and IsGrabbed(Camlock.Target) then return end
    local HitPosition = GetCamlockHitPosition(Camlock.Target)
    if not HitPosition then return end
    local Smoothing = _G.CamlockSmoothness
    local EasedSmoothing = TweenService:GetValue(Smoothing, Enum.EasingStyle[_G.CamlockEasingStyle] or Enum.EasingStyle.Quad, Enum.EasingDirection[_G.CamlockEasingDirection] or Enum.EasingDirection.Out)
    cam.CFrame = cam.CFrame:Lerp(CFrame.new(cam.CFrame.Position, HitPosition), EasedSmoothing)
end

function EnableCamlock()
    if not _G.CamlockEnabled then return end
    local target = GetBestCamlockTarget()
    if target then
        Camlock.Target = target
        Camlock.Active = true
        if not Camlock.Connection then
            Camlock.Connection = RunService.RenderStepped:Connect(UpdateCamlock)
        end
    end
end

function DisableCamlock()
    Camlock.Active = false
    Camlock.Target = nil
    if isMobile and mobileLockBtn then
        mobileLockBtn.BackgroundColor3 = medBrown
        mobileLockBtn.Text = "🍮"
    end
end

if not Camlock.Connection then
    Camlock.Connection = RunService.RenderStepped:Connect(UpdateCamlock)
end

        if _G.CamlockEnabled and not _G.CamlockAutoToggle then
            local camlockKey = getKeyCode(_G.CamlockToggleKey)
            if input.KeyCode == camlockKey then
                if _G.CamlockMode == "Toggle" then
                    if Camlock.Active then DisableCamlock() else EnableCamlock() end
                elseif _G.CamlockMode == "Hold" then EnableCamlock() end
            end
        end
        if _G.CamlockEnabled and _G.CamlockUseRMB and not _G.CamlockAutoToggle then
            if input.UserInputType == Enum.UserInputType.MouseButton2 then
                if _G.CamlockMode == "Toggle" then
                    if Camlock.Active then DisableCamlock() else EnableCamlock() end
                elseif _G.CamlockMode == "Hold" then EnableCamlock() end
            end
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if _G.CamlockEnabled and not _G.CamlockAutoToggle then
            local camlockKey = getKeyCode(_G.CamlockToggleKey)
            if input.KeyCode == camlockKey and _G.CamlockMode == "Hold" then DisableCamlock() end
        end
        if _G.CamlockEnabled and _G.CamlockUseRMB and not _G.CamlockAutoToggle then
            if input.UserInputType == Enum.UserInputType.MouseButton2 and _G.CamlockMode == "Hold" then DisableCamlock() end
        end
    end)
end

Players.PlayerRemoving:Connect(function(p)
    if Camlock.Target == p then DisableCamlock() end
end)

Player.CharacterAdded:Connect(function()
    DisableCamlock()
end)

-- Zoom logic
local targetZoom = _G.ZoomMax
task.spawn(function()
    while true do
        if _G.ZoomRateEnabled then
            if _G.ZoomOnlyWhenCamlocked and not Camlock.Active then
                task.wait(0.25)
            else
                if targetZoom >= _G.ZoomMax then
                    targetZoom = _G.ZoomMin
                    task.wait(math.max(_G.ZoomStayMin / math.max(_G.ZoomFrequency, 0.1), 0.05))
                else
                    targetZoom = _G.ZoomMax
                    task.wait(math.max(_G.ZoomStayMax / math.max(_G.ZoomFrequency, 0.1), 0.05))
                end
            end
        else
            task.wait(0.25)
        end
    end
end)

RunService.RenderStepped:Connect(function(dt)
    if not _G.ZoomRateEnabled then return end
    if _G.ZoomOnlyWhenCamlocked and not Camlock.Active then return end
    local currentMin = Player.CameraMinZoomDistance
    local currentMax = Player.CameraMaxZoomDistance
    local currentTarget = math.clamp(targetZoom, _G.ZoomMin, _G.ZoomMax)
    Player.CameraMinZoomDistance = currentMin + (currentTarget - currentMin) * math.clamp(dt * _G.ZoomSpeed, 0, 1)
    Player.CameraMaxZoomDistance = currentMax + (currentTarget - currentMax) * math.clamp(dt * _G.ZoomSpeed, 0, 1)
end)

-- Drag UI
local dragging = false
local dragStartInput, startFramePos

cardContainer.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        local mousePos = UserInputService:GetMouseLocation()
        local guiObjects = PlayerGui:GetGuiObjectsAtPosition(mousePos.X, mousePos.Y)
        local overInteractive = false
        for _, obj in ipairs(guiObjects) do
            if obj:IsA("TextButton") or obj:IsA("ImageButton") or obj:IsA("ScrollingFrame") then
                overInteractive = true
                break
            end
        end
        if not overInteractive then
            dragging = true
            dragStartInput = input.Position
            startFramePos = cardContainer.Position
        end
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStartInput
        cardContainer.Position = UDim2.new(startFramePos.X.Scale, startFramePos.X.Offset + delta.X, startFramePos.Y.Scale, startFramePos.Y.Offset + delta.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

    if input.KeyCode == Enum.KeyCode.RightShift then
        screenGui.Enabled = not screenGui.Enabled
    end
end)
